    # ③ 보안등급이 '공개'가 아니면 → AI 호출 전에 중단
    시트 = pd.read_excel("data/온라인쇼핑동향_2026_01.xlsx", sheet_name=None, dtype={"연월": str, "공표일자": str})

    호출.clear()
    try:

    except PermissionError as e:

    assert 호출 == [], "차단되기 전에 AI 함수가 호출됨"
    try:
        os.remove("data/보안등급_테스트.xlsx")
    except PermissionError:
        print("참고: 테스트 엑셀이 아직 열려 있어 지우지 못함 — load_input이 ExcelFile을 with 문으로 닫는지 확인(7차시)")
finally:
    if os.path.exists(백업):
        shutil.move(백업, "mywork/press_data.json")
    for 경로 in 만든_파일:
        if os.path.exists(경로):
            os.remove(경로)
print("Ch10 OK — 파이프라인 구조·검토필요 분기·보안등급 차단 검증 통과 (실제 실행은 8절)")

## 10. 문제 해결

| 증상 | 원인 | 조치 |
| --- | --- | --- |
| `ModuleNotFoundError: No module named 'mywork'` | `python mywork\main.py`로 실행함 | 프로젝트 폴더에서 **`python -m mywork.main`**으로 실행 |
| `검증 중에 실제 API를 호출하려 함` | main.py가 모듈 이름으로 함수를 불러 가짜 함수가 끼워지지 않음 | "main.py에서 함수를 from mywork.xxx import 형태로 가져오도록 고쳐줘" 요청 |
| `대외비 자료가 차단되지 않음` / `차단되기 전에 AI 함수가 호출됨` | 보안 확인 누락 또는 순서 오류 | "clean_and_structure 직후, AI 호출 전에 check_data_grade를 부르도록" 요청 |
| hwpx·pdf 파일이 없음 | export.py 미복사, 또는 PDF 한글 폰트 없음 | 2절 복사 확인. 로그의 `pdf 저장 실패` 메시지 확인(`.env`에 `KOREAN_FONT_PATH=한글 .ttf 경로`) |
| 로그의 한글이 깨짐 | `encoding="utf-8"` 누락 | "logging.basicConfig에 encoding='utf-8'을 넣어줘" 요청 |
| 실제 실행에서 `AuthenticationError` | `.env` 키 없음 | 4차시 1-1 확인 |
| 실제 실행이 `검토필요` | AI 문장에 근거 없는 숫자 | 정상적인 안전장치 동작. 불일치 숫자를 확인하고 다시 실행하거나 사람이 수정 |
